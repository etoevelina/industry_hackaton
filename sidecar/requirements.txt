# Зависимости сайдкара. Python 3.12, macOS arm64 — основная платформа.
# Установка по секциям, а не всё сразу:
#   python3 -m pip install -r sidecar/requirements.txt            # поставит всё
#   pip install websockets numpy opencv-python psutil             # минимум для запуска
#
# ВАЖНО: обязательна только секция CORE. Всё остальное опционально —
# отсутствующий пакет выключает свой канал (available() == False), сайдкар
# стартует, честно сообщает это в hello и работает в деградированном режиме.

# ---------------------------------------------------------------------------
# CORE — ОБЯЗАТЕЛЬНО. Без этого сайдкар не поднимется.
# websockets 15 — современный asyncio-сервер (websockets.asyncio.server).
# numpy <2 — совместимость с opencv/mediapipe/onnxruntime сборками под arm64.
# opencv-python нужен для камеры; без него работает только --headless/--mock.
# ---------------------------------------------------------------------------
websockets>=12.0,<16.0
numpy>=1.26,<2.0
opencv-python>=4.9,<5.0
psutil>=5.9,<8.0

# ---------------------------------------------------------------------------
# VISION — опционально (каналы vision, gaze).
# mediapipe: FaceMesh, поза головы, взгляд, мимика. Колёса под macOS arm64
# есть с 0.10.9. Нет mediapipe -> нет gaze/лица, остальное работает.
# ultralytics: YOLOv8n для телефона и посторонних предметов (вес в models/).
# onnxruntime: инференс лёгких моделей без torch.
# ---------------------------------------------------------------------------
# ВЕРХНЯЯ ГРАНИЦА КРИТИЧНА. Начиная с 0.10.22 из пакета вырезан legacy-API
# mediapipe.solutions, на котором построен FaceAnalyzer: в 0.10.35 и 1.x
# модуль отдаёт только Image/ImageFormat/tasks, и канал взгляда (пункт 2.2 ТЗ)
# молча уходит в available()==False. Проверено: 0.10.21 — последняя рабочая.
# Переход на Tasks API (FaceLandmarker) — отдельная задача, см. memory/backlog.md.
mediapipe>=0.10.9,<0.10.22 ; platform_system != "Linux" or platform_machine == "x86_64"
ultralytics>=8.1,<8.4
onnxruntime>=1.17,<2.0

# ---------------------------------------------------------------------------
# IDENTITY — опционально (канал identity).
# insightface + onnxruntime: ArcFace-эмбеддинги для сверки «тот же человек».
# Модели кладутся локально в models/, скачивание в рантайме запрещено.
# insightface собирается из исходников, требует компилятор — ставить отдельно.
# ---------------------------------------------------------------------------
insightface>=0.7.3,<0.8
scikit-image>=0.22,<0.26          # зависимость insightface, фиксируем явно

# ---------------------------------------------------------------------------
# AUDIO — опционально (канал audio).
# sounddevice: захват с микрофона (PortAudio; на macOS ставится из колеса).
# webrtcvad: детектор речи, дёшево и надёжно.
# librosa: MFCC для сравнения голоса с образцом владельца.
# ---------------------------------------------------------------------------
sounddevice>=0.4.6,<0.6
webrtcvad-wheels>=2.0.11,<3.0     # колесо вместо webrtcvad: не требует сборки
librosa>=0.10.1,<0.12

# ---------------------------------------------------------------------------
# REPORT — опционально, но нужно для выгрузки отчёта.
# jinja2: HTML-шаблон отчёта. cryptography: подпись отчёта (Ed25519)
# и проверка hash-chain доказательной базы.
# ---------------------------------------------------------------------------
jinja2>=3.1,<4.0
cryptography>=42.0,<47.0

# ──────────────────────────────────────────────────────────────────────────
# ВНИМАНИЕ: onnx НЕ СТАВИТЬ В РАБОЧЕЕ ОКРУЖЕНИЕ.
# Он нужен ТОЛЬКО для разового экспорта модели (scripts/fetch_models.sh) и
# тянет protobuf>=6.31, тогда как mediapipe требует protobuf<5. Установка onnx
# рядом с mediapipe ломает канал взгляда с ошибкой
#   AttributeError: 'MessageFactory' object has no attribute 'GetPrototype'
# Порядок действий при подготовке модели:
#   python3 -m venv .venv-export && .venv-export/bin/pip install ultralytics onnx
#   .venv-export/bin/python -c "from ultralytics import YOLO; YOLO('yolov8n.pt').export(format='onnx', imgsz=640, opset=12)"
# либо, если экспортировали в основном окружении, вернуть protobuf обратно:
#   pip install "protobuf>=4.25.3,<5"
# В рантайме onnx не нужен: модель читает onnxruntime, который protobuf не использует.
# ──────────────────────────────────────────────────────────────────────────
